 $php = 'C:\xampp\php\php.exe'
 $base = 'C:\xampp\htdocs\Plataforma-Servicio-TI'

& $php -v    # debe mostrar "PHP 8.2.12 (cli)" — si da error, revisa la ruta

Get-ChildItem "$base\api" -Recurse -Filter *.php | ForEach-Object {
    $out = & $php -l $_.FullName 2>&1
    if ($out -match 'No syntax errors') { Write-Host "OK: $($_.Name)" -ForegroundColor Green }
    else { Write-Host "ERROR: $($_.Name) -> $out" -ForegroundColor Red }
}
Get-ChildItem "$base\core" -Filter *.php | ForEach-Object {
    $out = & $php -l $_.FullName 2>&1
    if ($out -match 'No syntax errors') { Write-Host "OK: core\$($_.Name)" -ForegroundColor Green }
    else { Write-Host "ERROR: core\$($_.Name) -> $out" -ForegroundColor Red }
}
